# Configurations of the Go-vs-C comparison (reference/bench/compare.sh).
# Format: <id> <input file> <fpgrowth options...>
# Inputs are written by `go run ./bench/gen -all -dir <data>`; thresholds
# are chosen so that every run reports 10^4 to 10^6 patterns (except the
# M5 rows). A trailing "*" on the id marks a size-limited closed/maximal
# configuration where C may report more sets than Go (deviation M5, bug
# B5); its counts are checked per program (Go's never above C's).

# sparse: 20,000 transactions, 1,000 items, average length 9.3
sparse-s        sparse.tab        -s0.1
sparse-c        sparse.tab        -tc -s0.1
sparse-m        sparse.tab        -tm -s0.1
sparse-g        sparse.tab        -tg -s0.1
sparse-r        sparse.tab        -tr -s0.1 -c60
sparse-r-lift   sparse.tab        -tr -s0.05 -c50 -el -d150

# sparse-large: 200,000 transactions, 20,000 items, average length 11.1
large-s         sparse-large.tab  -s0.05
large-c         sparse-large.tab  -tc -s0.05
large-m         sparse-large.tab  -tm -s0.05
# size-limited closed/maximal (deviation M5): complete extension check
large-c-n2*     sparse-large.tab  -tc -s0.05 -n2
large-c-n3*     sparse-large.tab  -tc -s0.05 -n3
large-m-n2*     sparse-large.tab  -tm -s0.05 -n2
large-m-n3*     sparse-large.tab  -tm -s0.05 -n3

# dense: 20,000 transactions, 80 items, average length 28.1 (planted patterns)
dense-s         dense.tab         -s5
dense-c         dense.tab         -tc -s5
dense-m         dense.tab         -tm -s5
dense-g         dense.tab         -tg -s10
dense-r         dense.tab         -tr -s10 -c90 -n4
dense-r-lift    dense.tab         -tr -s10 -c80 -n4 -el -d150

# variants on dense data (frequent and closed item sets)
dense-s-As-l0   dense.tab         -s10 -As -l0
dense-s-As      dense.tab         -s10 -As
dense-s-Ac-j    dense.tab         -s10 -Ac -j
dense-s-Ac      dense.tab         -s10 -Ac
dense-s-Ad      dense.tab         -s10 -Ad
dense-s-At      dense.tab         -s10 -At
dense-c-As-l0   dense.tab         -tc -s10 -As -l0
dense-c-As      dense.tab         -tc -s10 -As
dense-c-Ac-j    dense.tab         -tc -s10 -Ac -j
dense-c-Ac      dense.tab         -tc -s10 -Ac
dense-c-Ad      dense.tab         -tc -s10 -Ad
dense-c-At      dense.tab         -tc -s10 -At

# long: 10,000 transactions, 400 items, average length 62.2
long-s          long.tab          -s2.5
long-c          long.tab          -tc -s2.5
long-m          long.tab          -tm -s2.5

# weighted: like sparse, integer weights 1..20 in the last field
weighted-s      weighted.tab      -w -s0.2
weighted-c      weighted.tab      -w -tc -s0.2

# M5: size-limited closed/maximal sets on independent dense items
# (docs/m5-investigation.md §7; dNxKdP: N transactions, K items, density P %)
m5-d2000-c-n1*  d2000x40d60.tab   -tc -s2 -n1
m5-d2000-c-n2*  d2000x40d60.tab   -tc -s2 -n2
m5-d2000-m-n2*  d2000x40d60.tab   -tm -s2 -n2
m5-d2000-c-n3*  d2000x40d60.tab   -tc -s2 -n3
m5-d2000-m-n3*  d2000x40d60.tab   -tm -s2 -n3
m5-d2000-c-n4*  d2000x40d60.tab   -tc -s2 -n4
m5-d1000-c-n2*  d1000x30d80.tab   -tc -s10 -n2
m5-d1000-m-n2*  d1000x30d80.tab   -tm -s10 -n2
m5-d1000-c      d1000x30d80.tab   -tc -s30
m5-d5000-c-n3*  d5000x60d40.tab   -tc -s5 -n3
m5-d5000-m-n3*  d5000x60d40.tab   -tm -s5 -n3
# Dense maximal sets of exactly the maximum size (independent audit
# M5A-1, issue #19): every 3-set of d20000x200d30 is frequent at -s2 and
# no 4-set is, so -tm -n3 reports all C(200,3) = 1,313,400 3-sets, each
# after a check that counts its whole cover. Slow (Go about 100 s, C
# about 40 s per -n3 run); select the other M5 rows with an id filter,
# e.g. '^(m5-d[125]000-|m5-tier-|m5-split-|large-[cm]-n)'.
m5-d20000-c-n2*    d20000x200d30.tab -tc -s2 -n2
m5-d20000-m-n2*    d20000x200d30.tab -tm -s2 -n2
m5-d20000-c-n3*    d20000x200d30.tab -tc -s2 -n3
m5-d20000-m-n3*    d20000x200d30.tab -tm -s2 -n3
m5-d20000-m-n3-As* d20000x200d30.tab -tm -s2 -n3 -As
m5-d20000-m-n3-At* d20000x200d30.tab -tm -s2 -n3 -At
m5-d20000-m-n4*    d20000x200d30.tab -tm -s1 -n4
# smin 2 transactions: every check stops at its first frequent extension,
# so sibling rows (issue #19) must not be built (design review F1)
m5-d20000-m-n2-s2* d20000x200d30.tab -tm -s-2 -n2
# Two-tier data (issue #19, performance review F1): each pair of rare
# items is a parent with 4 maximal children; rows built for them would
# pay off for at most one child, so they must not be built (no
# regression against main).
m5-tier-m-n3*      t100000x204c4.tab -tm -s-180 -n3
m5-tier-m-n3-As*   t100000x204c4.tab -tm -s-180 -n3 -As
m5-tier-m-n3-At*   t100000x204c4.tab -tm -s-180 -n3 -At
m5-tier-c-n3*      t100000x204c4.tab -tc -s-180 -n3
# Slices and spreaders (issue #19, performance review R2-1): most
# children of a rare item are never checked (they have frequent
# supersets), so the later-children bound must not be trusted blindly.
m5-split-m-n3*     s50000x414.tab    -tm -s-1350 -n3
m5-split-m-n3-As*  s50000x414.tab    -tm -s-1350 -n3 -As
m5-split-m-n3-At*  s50000x414.tab    -tm -s-1350 -n3 -At
